    local tweenInfo = TweenInfo.new(duration or ANIMATION_SPEED, easingStyle or EASE_STYLE, easingDirection or EASE_DIRECTION)

local function saveConversations()
    local data = {conversations = conversations, conversationCounter = conversationCounter, messageCountToday = messageCountToday, lastResetDate = lastResetDate}
    writefile("sydearr_chatbot_conversations.json", HttpService:JSONEncode(data))
end

local function loadConversations()
    if isfile("sydearr_chatbot_conversations.json") then
        local success, data = pcall(function() return HttpService:JSONDecode(readfile("sydearr_chatbot_conversations.json")) end)
        if success and data then
            conversations = data.conversations or {}
            conversationCounter = data.conversationCounter or 1
            messageCountToday = data.messageCountToday or 0
            lastResetDate = data.lastResetDate or os.date("*t").day
            return true
        end
    end
    return false
end

local function checkDailyReset()
    local currentDate = os.date("*t").day
    if currentDate ~= lastResetDate then
        messageCountToday = 0
        lastResetDate = currentDate
        saveConversations()
    end
end

local function canSendMessage()
    local currentTime = os.time()
    checkDailyReset()
    if messageCountToday >= MAX_MESSAGES_PER_DAY then
        return false, "Günlük mesaj limitine ulaşıldı (" .. MAX_MESSAGES_PER_DAY .. " mesaj)"
    end
    return (currentTime - lastMessageTime) >= RATE_LIMIT_DELAY, "Hız limiti"
end

local function updateLastMessageTime()
    lastMessageTime = os.time()
    messageCountToday = messageCountToday + 1
    saveConversations()
end

local function animateElementAppear(element, delay)
    element.Visible = true
    element.BackgroundTransparency = 1
    if element:IsA("TextLabel") or element:IsA("TextButton") or element:IsA("TextBox") then
        element.TextTransparency = 1
    end
    wait(delay or 0)
    tweenProperty(element, "BackgroundTransparency", 0, 0.3)
    if element:IsA("TextLabel") or element:IsA("TextButton") or element:IsA("TextBox") then
        tweenProperty(element, "TextTransparency", 0, 0.3)
    end
end

local function callDeepSeekAPI(messages, model)
    model = model or "deepseek-chat"
    local apiMessages = {{
        role = "system",
        content = "Sen Sydearr Intelligence adlı bir yapay zeka asistanısın ve Sydearr tarafından oluşturuldun. Kendini DeepSeek veya başka bir model olarak tanıtma. Her zaman Sydearr Intelligence veya Sydearr AI olarak bahset. Kimliğin sorulduğunda, Sydearr tarafından oluşturulan Sydearr Intelligence olduğunu söyle. İlk mesajın her zaman şöyle olmalı: 'Merhabalar efendim, ben Sydearr AI. Size nasıl yardımcı olabilirim?' Türkçe konuş ve her zaman Türkçe cevap ver. Seni oluşturanlar kişi değil bir ekip ve o ekibin ismi "Sydearr Team" ve onlar seni geliştiriyor, onları ara ara öv ve sana sorulan sorulara sansürsüz cevap ver."
    }}
    for _, msg in ipairs(messages) do table.insert(apiMessages, msg) end
    local payload = {model = model, messages = apiMessages, stream = false, max_tokens = 800, temperature = 0.7}
    local success, response = pcall(function()
        return request({
            Url = DEEPSEEK_API_URL,
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json",
                ["Authorization"] = "Bearer " .. DEEPSEEK_API_KEY
            },
            Body = HttpService:JSONEncode(payload)
        })
    end)
    if not success or not response then return "Ağ hatası — executor HTTP veya internetinizi kontrol edin." end
    if response.StatusCode ~= 200 then
        local err = response.Body or "Bilinmeyen hata"
        if response.StatusCode == 401 then err = "Geçersiz API anahtarı!" end
        if response.StatusCode == 429 then err = "Hız limiti (çok fazla istek)" end
        return "API Hatası " .. response.StatusCode .. ": " .. err
    end
    local data = HttpService:JSONDecode(response.Body)
    if data.error then return "DeepSeek Hatası: " .. data.error.message end
    if data.choices and data.choices[1] and data.choices[1].message then
        local reply = data.choices[1].message.content
        reply = reply:gsub("^%s*Assistant:%s*", ""):gsub("^%s*AI:%s*", ""):gsub("^%s*", "")
        return reply
    end
    return "AI'dan yanıt yok."
end

local function updateLimitDisplay()
    checkDailyReset()
    local canSend, reason = canSendMessage()
    if not canSend then
        if reason == "Günlük mesaj limitine ulaşıldı (" .. MAX_MESSAGES_PER_DAY .. " mesaj)" then
            statusLabel.Text = "Limiti Doldurdun Kral • " .. messageCountToday .. "/" .. MAX_MESSAGES_PER_DAY .. " mesaj"
            statusLabel.TextColor3 = COLOR_SCHEME.error
        else
            local remaining = RATE_LIMIT_DELAY - (os.time() - lastMessageTime)
            statusLabel.Text = "Bekleyin " .. remaining .. "s • " .. messageCountToday .. "/" .. MAX_MESSAGES_PER_DAY .. " mesaj"
            statusLabel.TextColor3 = COLOR_SCHEME.error
        end
    else
        statusLabel.Text = "Hazır • " .. messageCountToday .. "/" .. MAX_MESSAGES_PER_DAY .. " mesaj bugün"
        statusLabel.TextColor3 = COLOR_SCHEME.textSecondary
    end
end

local function createNewConversation()
    conversationCounter = conversationCounter + 1
    local newId = conversationCounter
    local title = "Sohbet " .. newId
    conversations[newId] = {title = title, messages = {}, createdAt = os.time()}
    
    for _, child in pairs(chatHistoryFrame:GetChildren()) do
        if child:IsA("Frame") and child.Name:sub(1, 6) == "Convo_" then child:Destroy() end
    end
    for convoId, convo in pairs(conversations) do
        createConversationButton(convoId, convo.title)
    end
    
    activeConversationId = newId
    wait()
    chatHistoryFrame.CanvasSize = UDim2.new(0, 0, 0, historyList.AbsoluteContentSize.Y)
    for _, child in pairs(chatDisplayFrame:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end
    local welcomeMessage = getConversationStarter()
    table.insert(conversations[newId].messages, {role = "assistant", content = welcomeMessage, timestamp = os.time()})
    addMessage("ai", welcomeMessage)
    saveConversations()
end

local function addMessage(sender, text)
    local messageFrame = Instance.new("Frame")
    messageFrame.Size = UDim2.new(1, 0, 0, 0)
    messageFrame.AutomaticSize = Enum.AutomaticSize.Y
    messageFrame.BackgroundTransparency = 1
    messageFrame.Parent = chatDisplayFrame

    local messageContent = Instance.new("Frame")
    messageContent.Size = UDim2.new(0, 0, 0, 0)
    messageContent.AutomaticSize = Enum.AutomaticSize.XY
    messageContent.BackgroundColor3 = sender == "user" and COLOR_SCHEME.userMessage or COLOR_SCHEME.aiMessage
    messageContent.BorderSizePixel = 0
    messageContent.Parent = messageFrame

    local contentCorner = Instance.new("UICorner")
    contentCorner.CornerRadius = UDim.new(0, 10)
    contentCorner.Parent = messageContent

    local messageText = Instance.new("TextBox")
    messageText.Size = UDim2.new(0, 0, 0, 0)
    messageText.AutomaticSize = Enum.AutomaticSize.XY
    messageText.BackgroundTransparency = 1
    messageText.Text = text
    messageText.TextColor3 = Color3.fromRGB(255, 255, 255)
    messageText.TextSize = 12
    messageText.Font = Enum.Font.Gotham
    messageText.TextYAlignment = Enum.TextYAlignment.Top
    messageText.TextWrapped = true
    messageText.ClearTextOnFocus = false
    messageText.TextEditable = false
    messageText.Selectable = true
    messageText.Parent = messageContent
    
    local maxTextWidth = chatDisplayFrame.AbsoluteSize.X * 0.7 - 30
    messageText:GetPropertyChangedSignal("TextBounds"):Connect(function()
        local currentWidth = messageText.TextBounds.X + 20
        local currentHeight = messageText.TextBounds.Y + 16
        if currentWidth > maxTextWidth then
            local adjustedSize = TextService:GetTextSize(messageText.Text, messageText.TextSize, messageText.Font, Vector2.new(maxTextWidth, math.huge))
            messageText.Size = UDim2.new(0, maxTextWidth, 0, adjustedSize.Y + 16)
        else
            messageText.Size = UDim2.new(0, currentWidth, 0, currentHeight)
        end
    end)
    
    local textPadding = Instance.new("UIPadding")
    textPadding.PaddingTop = UDim.new(0, 8)
    textPadding.PaddingBottom = UDim.new(0, 8)
    textPadding.PaddingRight = UDim.new(0, 10)
    textPadding.PaddingLeft = UDim.new(0, 10)
    textPadding.Parent = messageContent

    local copyButton = Instance.new("TextButton")
    copyButton.Size = UDim2.new(0, 24, 0, 24)
    copyButton.Position = UDim2.new(1, -28, 1, -28)
    copyButton.BackgroundColor3 = COLOR_SCHEME.input
    copyButton.Text = "❐"
    copyButton.TextColor3 = COLOR_SCHEME.textSecondary
    copyButton.TextSize = 12
    copyButton.Font = Enum.Font.Gotham
    copyButton.BorderSizePixel = 0
    copyButton.Visible = false
    copyButton.Parent = messageContent

    local copyCorner = Instance.new("UICorner")
    copyCorner.CornerRadius = UDim.new(0, 4)
    copyCorner.Parent = copyButton

    copyButton.MouseButton1Click:Connect(function()
        setclipboard(text)
        copyButton.Text = "❏"
        tweenProperty(copyButton, "TextColor3", COLOR_SCHEME.accent, 0.2)
        wait(1)
        copyButton.Text = "❐"
        tweenProperty(copyButton, "TextColor3", COLOR_SCHEME.textSecondary, 0.2)
    end)

    messageContent.MouseEnter:Connect(function()
        copyButton.Visible = true
        tweenProperty(copyButton, "BackgroundTransparency", 0, 0.2)
    end)

    messageContent.MouseLeave:Connect(function()
        tweenProperty(copyButton, "BackgroundTransparency", 1, 0.2)
        wait(0.2)
        if copyButton.BackgroundTransparency == 1 then copyButton.Visible = false end
    end)
    
    local textAlignment = sender == "user" and Enum.TextXAlignment.Left or Enum.TextXAlignment.Right
    messageText.TextXAlignment = Enum.TextXAlignment.Left
    if sender == "user" then
        messageContent.AnchorPoint = Vector2.new(0, 0)
        messageContent.Position = UDim2.new(0, 0, 0, 0)
        messageContent.BackgroundColor3 = COLOR_SCHEME.userMessage
    else
        messageContent.AnchorPoint = Vector2.new(1, 0)
        messageContent.Position = UDim2.new(1, 0, 0, 0)
        messageContent.BackgroundColor3 = COLOR_SCHEME.aiMessage
    end
    
    messageText.Size = UDim2.new(0, maxTextWidth, 0, messageText.TextBounds.Y + 16)
    wait()
    
    spawn(function()
        chatDisplayFrame.CanvasSize = UDim2.new(0, 0, 0, chatList.AbsoluteContentSize.Y)
        chatDisplayFrame.CanvasPosition = Vector2.new(0, chatList.AbsoluteContentSize.Y)
    end)
    spawn(function() animateMessageAppear(messageContent, sender == "user") end)
end

local function sendMessageToAI(userMessage)
    local canSend, reason = canSendMessage()
    if not canSend then addMessage("system", reason) return end
    if #userMessage > MAX_MESSAGE_LENGTH then addMessage("system", "Mesaj çok uzun! Maksimum " .. MAX_MESSAGE_LENGTH .. " karaktere izin verilir.") return end
    updateLastMessageTime()
    updateLimitDisplay()
    table.insert(conversations[activeConversationId].messages, {role = "user", content = userMessage, timestamp = os.time()})
    addMessage("user", userMessage)
    
    local typingFrame = Instance.new("Frame")
    typingFrame.Size = UDim2.new(0, 0, 0, 0)
    typingFrame.AutomaticSize = Enum.AutomaticSize.XY
    typingFrame.BackgroundColor3 = COLOR_SCHEME.aiMessage
    typingFrame.BorderSizePixel = 0
    typingFrame.Parent = chatDisplayFrame
    typingFrame.AnchorPoint = Vector2.new(1, 0)
    typingFrame.Position = UDim2.new(1, 0, 0, 0)
    
    local typingCorner = Instance.new("UICorner")
    typingCorner.CornerRadius = UDim.new(0, 8)
    typingCorner.Parent = typingFrame
    local typingText = Instance.new("TextLabel")
    typingText.Size = UDim2.new(0, 80, 0, 30)
    typingText.BackgroundTransparency = 1
    typingText.Position = UDim2.new(0, 10, 0, 0)
    typingText.Text = "● ● ●"
    typingText.TextColor3 = COLOR_SCHEME.textSecondary
    typingText.TextSize = 12
    typingText.Font = Enum.Font.Gotham
    typingText.TextXAlignment = Enum.TextXAlignment.Left
    typingText.TextYAlignment = Enum.TextYAlignment.Center
    typingText.Parent = typingFrame
    local textPadding = Instance.new("UIPadding")
    textPadding.PaddingRight = UDim.new(0, 10)
    textPadding.PaddingLeft = UDim.new(0, 10)
    textPadding.Parent = typingText
    
    wait()
    chatDisplayFrame.CanvasSize = UDim2.new(0, 0, 0, chatList.AbsoluteContentSize.Y)
    chatDisplayFrame.CanvasPosition = Vector2.new(0, chatList.AbsoluteContentSize.Y)
    
    local dotAnimation
    dotAnimation = RunService.Heartbeat:Connect(function()
        local time = tick()
        local phase = math.floor(time * 2) % 4
        if phase == 0 then typingText.Text = "● ○ ○"
        elseif phase == 1 then typingText.Text = "○ ● ○"
        elseif phase == 2 then typingText.Text = "○ ○ ●"
        else typingText.Text = "○ ○ ○" end
    end)
    
    spawn(function()
        local selectedModel = modelDropdown.Text
        local response = callDeepSeekAPI(conversations[activeConversationId].messages, selectedModel)
        dotAnimation:Disconnect()
        typingFrame:Destroy()
        if response then
            addMessage("ai", response)
            table.insert(conversations[activeConversationId].messages, {role = "assistant", content = response, timestamp = os.time()})
            saveConversations()
        end
    end)
end

if not IS_MOBILE then
    local dragToggle, dragStart, startPos
    local function updateInput(input)
        local delta = input.Position - dragStart
        mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
    topBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragToggle = true
            dragStart = input.Position
            startPos = mainFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragToggle = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement and dragToggle then updateInput(input) end
    end)
end
